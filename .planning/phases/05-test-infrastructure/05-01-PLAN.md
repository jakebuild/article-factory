    **Fixture 1: `db_session` — isolated in-memory SQLite per test**

    The challenge: `database.py` initialises a global `_engine` and `SessionLocal` at import time pointing at a real file. The fixture must redirect those globals to an in-memory engine for the duration of each test, then restore them.

    ```python
    import pytest
    from sqlalchemy import create_engine
    from sqlalchemy.orm import sessionmaker
    from article_factory.models import Base, Topic, TopicStatus
    import article_factory.database as db_module

    @pytest.fixture
    def db_session(monkeypatch):
        """Isolated in-memory SQLite session pre-seeded with two topics."""
        # Create a fresh in-memory engine each test
        engine = create_engine(
            "sqlite:///:memory:",
            connect_args={"check_same_thread": False},
        )
        Base.metadata.create_all(engine)
        Session = sessionmaker(bind=engine)

        # Patch the database module globals so all db_module functions
        # (create_topic, get_topic, update_status, etc.) use the in-memory DB
        monkeypatch.setattr(db_module, "_engine", engine)
        monkeypatch.setattr(db_module, "SessionLocal", Session)

        # Seed two topics so tests have data without DB state bleed-through
        session = Session()
        t1 = Topic(topic="Python Async", prompt="Write about async Python", status=TopicStatus.NEW)
        t2 = Topic(topic="AI Safety", prompt="Write about AI safety", status=TopicStatus.PROCESSING)
        session.add_all([t1, t2])
        session.commit()
        session.close()

        # Yield a fresh session for tests that need direct session access
        session = Session()
        yield session
        session.close()
        engine.dispose()
    ```

    from article_factory.notebook_lm import NotebookLMClientWrapper

    @pytest.fixture

        generate_audio, wait_for_completion, download_audio, import_sources,
        list_sources, generate_infographic, download_infographic) are AsyncMocks.

        Use: inject as argument or use patch() in tests that instantiate the wrapper.
        """
        client = MagicMock(spec=NotebookLMClientWrapper)
        client.create_notebook = AsyncMock(return_value="notebook-test-id")
        client.get_notebook = AsyncMock(return_value={"id": "notebook-test-id", "title": "Test"})
        client.start_research = AsyncMock(return_value={"task_id": "research-task-id"})
        client.poll_research = AsyncMock(return_value={
            "status": "completed",
            "sources": [{"title": "Source 1", "url": "https://example.com"}],

        client.generate_audio = AsyncMock(return_value={"task_id": "audio-task-id"})
        client.wait_for_completion = AsyncMock(return_value={

        client.download_audio = AsyncMock(return_value="/tmp/audio.mp3")

        client.list_sources = AsyncMock(return_value=[{"id": "src-1", "title": "Source 1"}])
        client.generate_infographic = AsyncMock(return_value={"task_id": "infographic-task-id"})
        client.download_infographic = AsyncMock(return_value="/tmp/infographic.png")
